Test Cases
Test 1: Create Order from Cart Items
Method: POST
URL: {{base_url}}/orders
Headers:

text
Authorization: Bearer {{auth_token}}
Content-Type: application/json
Body (raw JSON):

json
{
  "cartIds": [1, 2]
}
Tests Script:

javascript
pm.test("Status code is 201", function () {
    pm.response.to.have.status(201);
});

pm.test("Response has correct structure", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData).to.have.property('status');
    pm.expect(jsonData).to.have.property('message');
    pm.expect(jsonData).to.have.property('order');
    pm.expect(jsonData.status).to.eql('success');
});

pm.test("Order details are correct", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.order).to.have.property('id');
    pm.expect(jsonData.order).to.have.property('customer_id');
    pm.expect(jsonData.order).to.have.property('total_amount');
    pm.expect(jsonData.order).to.have.property('status');
    pm.expect(jsonData.order.status).to.eql('pending');
    
    // Save order_id for future tests
    pm.environment.set("order_id", jsonData.order.id);
});

pm.test("Response time is less than 3000ms", function () {
    pm.expect(pm.response.responseTime).to.be.below(3000);
});

Test 5: Get Customer Orders
Method: GET
URL: {{base_url}}/orders/{{customer_id}}
Headers:

text
Authorization: Bearer {{auth_token}}
Tests Script:

javascript
pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

pm.test("Orders response structure", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData).to.have.property('status');
    pm.expect(jsonData).to.have.property('message');
    pm.expect(jsonData).to.have.property('orders');
    pm.expect(jsonData).to.have.property('summary');
    pm.expect(jsonData.status).to.eql('success');
});

pm.test("Orders array structure", function () {
    var jsonData = pm.response.json();
    if (jsonData.orders.length > 0) {
        var order = jsonData.orders[0];
        pm.expect(order).to.have.property('id');
        pm.expect(order).to.have.property('customer_id');
        pm.expect(order).to.have.property('total_amount');
        pm.expect(order).to.have.property('status');
        pm.expect(order).to.have.property('created_at');
        pm.expect(order).to.have.property('items');
    }
});

pm.test("Summary structure", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.summary).to.have.property('total_orders');
    pm.expect(jsonData.summary).to.have.property('total_spent');
    pm.expect(jsonData.summary.total_orders).to.be.a('number');
    pm.expect(jsonData.summary.total_spent).to.be.a('string');
});

javascript
// Check if authenticated
if (!pm.environment.get("auth_token")) {
    console.log("⚠️ No auth token. Please login first.");
}

// Log the request
console.log(`Making ${pm.request.method} request to: ${pm.request.url}`);
Test Script (Collection level):

javascript
// Log response details
console.log(`Response: ${pm.response.status} ${pm.response.reason()}`);
console.log(`Response time: ${pm.response.responseTime}ms`);

// Common tests for all endpoints
pm.test("Response has JSON content-type", function () {
    pm.expect(pm.response.headers.get('Content-Type')).to.include('application/json');
});

pm.test("Response is valid JSON", function () {
    pm.response.to.be.json;
});
